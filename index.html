var SHEET_NAME = 'Data'; function doGet(e) { var action = (e.parameter.action || '').toString(); if (action === 'read') return readData(); if (action === 'twprice') return getTWPrice(e.parameter.sym); if (action === 'ytd') return getYtd(e.parameter.sym, e.parameter.mkt, e.parameter.year); return jsonOut({ok:false}); } function doPost(e) { try { var d = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); sh.clear(); sh.getRange('A1').setValue(d.trades || '[]'); sh.getRange('B1').setValue(d.prices || '{}'); sh.getRange('C1').setValue(d.history || '[]'); return jsonOut({ok:true}); } catch(err) { return jsonOut({ok:false,e:err.message}); } } function readData() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) return jsonOut({ok:false,msg:'no sheet'}); var trades = sh.getRange('A1').getValue(); var prices = sh.getRange('B1').getValue(); var history = sh.getRange('C1').getValue(); return jsonOut({ok:true,trades:trades,prices:prices,history:history||'[]'}); } catch(err) { return jsonOut({ok:false,e:err.message}); } } function getTWPrice(sym) { try { var url = 'https://mis.twse.com.tw/stock/api/getStockInfo.jsp?ex_ch=tse_' + sym + '.tw'; var r = JSON.parse(UrlFetchApp.fetch(url).getContentText()); var d = r.msgArray && r.msgArray[0]; var p = d ? parseFloat(d.z) || parseFloat(d.y) : 0; var y = d ? parseFloat(d.y) : 0; return jsonOut({ok:true,price:p,yesterday:y}); } catch(err) { return jsonOut({ok:false,price:0}); } } /* Fetch year-start (first trading day) price via Yahoo Finance chart endpoint. Works for both US and TW symbols. For TW, .TW suffix is auto-appended. */ function getYtd(sym, mkt, year) { try { var y = parseInt(year) || (new Date()).getFullYear(); var ticker = sym; if (mkt === 'tw' && sym.indexOf('.') < 0) ticker = sym + '.TW'; // period1 = Jan 1 UTC, period2 = Jan 20 UTC (covers any market open day in early Jan) var p1 = Math.floor(Date.UTC(y, 0, 1) / 1000); var p2 = Math.floor(Date.UTC(y, 0, 20) / 1000); var url = 'https://query1.finance.yahoo.com/v8/finance/chart/' + encodeURIComponent(ticker) + '?period1=' + p1 + '&period2=' + p2 + '&interval=1d'; var opts = { method: 'get', muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0 (compatible; Portfolio/1.0)' } }; var resp = UrlFetchApp.fetch(url, opts); var code = resp.getResponseCode(); if (code !== 200) return jsonOut({ok:false, price:0, code:code, ticker:ticker}); var body = JSON.parse(resp.getContentText()); var result = body && body.chart && body.chart.result && body.chart.result[0]; if (!result) return jsonOut({ok:false, price:0, ticker:ticker, msg:'no data'}); var quote = result.indicators && result.indicators.quote && result.indicators.quote[0]; var opens = quote && quote.open; var closes = quote && quote.close; if (!opens || !opens.length) return jsonOut({ok:false, price:0, ticker:ticker, msg:'no opens'}); // First non-null open price (Yahoo sometimes pads weekends/holidays with null) var price = 0; for (var i = 0; i < opens.length; i++) { if (opens[i] != null && opens[i] > 0) { price = opens[i]; break; } } if (!price && closes) { for (var i = 0; i < closes.length; i++) { if (closes[i] != null && closes[i] > 0) { price = closes[i]; break; } } } return jsonOut({ok: price > 0, price: price, ticker: ticker}); } catch(err) { return jsonOut({ok:false, price:0, e:err.message}); } } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
